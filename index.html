async function fetch2DData() { try { // API လိပ်စာအမှန်ကို ဤနေရာတွင် ထည့်ပါ const apiUrl = 'https://your-api-domain.com/get_data'; const response = await fetch(apiUrl); const data = await response.json(); // ၁။ LIVE အပိုင်းကို ထည့်သွင်းခြင်း if (data.live) { document.getElementById('top-num').innerText = data.live.twod || "--"; document.getElementById('status-time').innerText = data.live.time || "--"; } // ၂။ RESULT အပိုင်း if (data.result && Array.isArray(data.result)) { data.result.forEach(item => { if (item.open_time === "09:30:00") { document.getElementById('m930').innerText = item.modern || "--"; document.getElementById('i930').innerText = item.internet || "--"; } if (item.open_time === "12:01:00") { document.getElementById('c1-set').innerText = item.set || "--"; document.getElementById('c1-val').innerText = item.value || "--"; document.getElementById('c1-num').innerText = item.twod || "--"; } if (item.open_time === "14:00:00") { document.getElementById('m200').innerText = item.modern || "--"; document.getElementById('i200').innerText = item.internet || "--"; } if (item.open_time === "16:30:00") { document.getElementById('c2-set').innerText = item.set || "--"; document.getElementById('c2-val').innerText = item.value || "--"; document.getElementById('c2-num').innerText = item.twod || "--"; } }); } } catch (error) { console.error("API Fetching Error:", error); document.getElementById('status-time').innerText = "Connection Error"; } } fetch2DData(); setInterval(fetch2DData, 5000);
